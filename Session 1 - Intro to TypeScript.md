# Verify Setup Typescript    
    * tsc --version

 # Typescript Configuration
    * tsc -- init

 # Convert Typescript Files to Javascript Files
    * tsc   

 # Datatypes   
    * let x:string; => string
    * let x:number; => number (integer, float, decimal)
    * let x:boolean; => boolean (true , false)
    * let x:undefined; => undefined
    * let x:null; => null
    * let x:object; => object {}
    * let x:union; => variable contain more of one (number | string | boolean) => ممكن المتغير ان يحمل اكثر من نوع للبيانات
    * let x:tuple; => fixed array
    * let x:"value" | "value" => type literal بيحدد القيم المخزنة داخل المتغير
    * let x:any => contain any datatype
    * (custom type) type = name => contain one or more of datatypes and called alias name for uses
